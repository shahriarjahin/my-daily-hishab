cmake_minimum_required(VERSION 3.22.1)
project(hishab_ai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID_ABI STREQUAL "armeabi-v7a" OR ANDROID_ABI STREQUAL "x86")
  # Preserve bookkeeping on older 32-bit phones; the optional model requires 64-bit.
  add_library(hishab_ai SHARED unsupported_ai.cpp)
  return()
endif()
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
file(READ "${REPO}/mobile/offline-ai.json" CONFIG_JSON)
string(JSON ENGINE_COMMIT GET "${CONFIG_JSON}" engine_commit)
add_subdirectory("${REPO}/mobile/.tools/llama.cpp-${ENGINE_COMMIT}" llama)
add_library(hishab_ai SHARED offline_ai.cpp)
target_link_libraries(hishab_ai PRIVATE llama android log)
target_link_options(hishab_ai PRIVATE "-Wl,-z,max-page-size=16384")
